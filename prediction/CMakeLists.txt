cmake_minimum_required(VERSION 3.16)
project(armor_native_prediction LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(OpenCV 4 REQUIRED COMPONENTS core calib3d imgproc)
add_executable(armor-predict src/adapter.cpp src/coordinates.cpp src/pose_solver.cpp src/research_tracker.cpp src/target.cpp third_party/tongji/tasks/auto_aim/armor.cpp third_party/tongji/tools/extended_kalman_filter.cpp third_party/tongji/tools/math_tools.cpp)
target_include_directories(armor-predict PRIVATE include compat third_party/tongji ${OpenCV_INCLUDE_DIRS})
target_link_libraries(armor-predict PRIVATE Eigen3::Eigen ${OpenCV_LIBS})
target_compile_options(armor-predict PRIVATE -O2)
